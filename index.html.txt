            <button class="tab-btn" onclick="switchCategory('初中', this)">初中 (中一至中三)</button>

    <script>
        // 設定登入密碼
        const TEACHER_CODE = "ks23417422";
        let isLoggedIn = false;
        let currentCategory = '初小';

        // 年級與組別對應關係
        const categoryMap = {
            '初小': ['小一', '小二', '小三'],
            '高小': ['小四', '小五', '小六'],
            '初中': ['中一', '中二', '中三'],
            '高中': ['中四', '中五', '中六']
        };

        // 預設遊戲資料 (若無本地紀錄時載入)
        const defaultGames = [
            { id: 1, title: '趣味加減法', grade: '小一', url: 'https://example.com', img: 'https://picsum.photos/200/140?random=1' },
            { id: 2, title: '英文單字對對碰', grade: '小四', url: 'https://example.com', img: 'https://picsum.photos/200/140?random=2' },
            { id: 3, title: '科學元素挑戰', grade: '中二', url: 'https://example.com', img: 'https://picsum.photos/200/140?random=3' },
            { id: 4, title: '歷史知識問答', grade: '中五', url: 'https://example.com', img: 'https://picsum.photos/200/140?random=4' }
        ];

        // 讀取遊戲資料
        function getGames() {
            const saved = localStorage.getItem('gamesData');
            return saved ? JSON.parse(saved) : defaultGames;
        }

        // 儲存遊戲資料
        function saveGames(games) {
            localStorage.setItem('gamesData', JSON.stringify(games));
        }

        // 老師登入 / 登出 切換
        function toggleTeacherAuth() {
            if (isLoggedIn) {
                isLoggedIn = false;
                document.getElementById('authBtn').innerText = "老師登入";
                document.getElementById('adminPanel').classList.remove('active');
                alert("已成功登出老師模式");
            } else {
                const inputCode = prompt("請輸入老師登入碼：");
                if (inputCode === TEACHER_CODE) {
                    isLoggedIn = true;
                    document.getElementById('authBtn').innerText = "登出管理員";
                    document.getElementById('adminPanel').classList.add('active');
                    alert("登入成功！您可以新增或刪除遊戲連結。");
                } else if (inputCode !== null) {
                    alert("登入碼錯誤，請重新輸入！");
                }
            }
            renderGames(); // 重新渲染以顯示/隱藏刪除按鈕
        }

        // 切換分類頁籤
        function switchCategory(category, btnElement) {
            currentCategory = category;
            
            // 更新頁籤 active 樣式
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            btnElement.classList.add('active');

            renderGames();
        }

        // 新增遊戲
        function addGame(event) {
            event.preventDefault();
            
            const title = document.getElementById('gameTitle').value;
            const grade = document.getElementById('gameGrade').value;
            const url = document.getElementById('gameUrl').value;
            let img = document.getElementById('gameImg').value;

            if (!img) {
                img = 'https://picsum.photos/200/140?random=' + Date.now(); // 預設占位圖
            }

            const newGame = {
                id: Date.now(),
                title: title,
                grade: grade,
                url: url,
                img: img
            };

            const games = getGames();
            games.push(newGame);
            saveGames(games);

            // 清空表單
            document.getElementById('addGameForm').reset();
            alert("新增成功！");

            // 自動切換到對應的組別查看
            for (const [cat, grades] of Object.entries(categoryMap)) {
                if (grades.includes(grade)) {
                    const tabs = document.querySelectorAll('.tab-btn');
                    const targetBtn = Array.from(tabs).find(btn => btn.innerText.includes(cat));
                    if (targetBtn) switchCategory(cat, targetBtn);
                    break;
                }
            }
        }

        // 刪除遊戲
        function deleteGame(id) {
            if (confirm("確定要刪除這個遊戲連結嗎？")) {
                let games = getGames();
                games = games.filter(game => game.id !== id);
                saveGames(games);
                renderGames();
            }
        }

        // 渲染遊戲列表
        function renderGames() {
            const gamesContainer = document.getElementById('gamesContainer');
            gamesContainer.innerHTML = '';

            const games = getGames();
            const allowedGrades = categoryMap[currentCategory];

            // 篩選當前組別的遊戲
            const filteredGames = games.filter(game => allowedGrades.includes(game.grade));

            if (filteredGames.length === 0) {
                gamesContainer.innerHTML = `<div class="empty-msg">目前【${currentCategory}】分頁尚無遊戲連結。</div>`;
                return;
            }

            filteredGames.forEach(game => {
                const card = document.createElement('div');
                card.className = 'game-card';

                const deleteBtnHtml = isLoggedIn 
                    ? `<button class="delete-btn" onclick="deleteGame(${game.id})">🗑️ 刪除連結</button>` 
                    : '';

                card.innerHTML = `
                    <img src="${game.img}" alt="${game.title}" onerror="this.src='https://via.placeholder.com/200x140?text=No+Image'">
                    <div class="game-info">
                        <div>
                            <span class="game-grade">${game.grade}</span>
                            <div class="game-title">${game.title}</div>
                        </div>
                        <div>
                            <a href="${game.url}" target="_blank" class="play-btn">🎮 開始遊戲</a>
                            ${deleteBtnHtml}
                        </div>
                    </div>
                `;
                gamesContainer.appendChild(card);
            });
        }

        // 初始化頁面
        window.onload = function() {
            renderGames();
        };
    </script>
